Developer Code Examples (Publishing via API) 

curl -X POST "{{ request.scheme }}://{{ request.get_host }}/api/v1/posts/create/" \
  -H "Authorization: Bearer {{ api_key }}" \
  -H "Content-Type: application/json" \
  -d '{
    "platform": "INSTAGRAM",
    "caption": "Scheduled automatically via FeedBridge Developer API! 🚀 #automation #socialmedia",
    "media_url": "https://images.unsplash.com/photo-1618005182384-a83a8bd57fbe?w=1080"
  }'
import requests

url = "{{ request.scheme }}://{{ request.get_host }}/api/v1/posts/create/"
headers = {
    "Authorization": "Bearer {{ api_key }}",
    "Content-Type": "application/json"
}
payload = {
    "platform": "YOUTUBE",
    "title": "My Viral Video Title",
    "caption": "Check out this new tutorial! Links in the description.",
    "media_url": "https://example.com/sample_video.mp4"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())
const response = await fetch("{{ request.scheme }}://{{ request.get_host }}/api/v1/posts/create/", {
  method: "POST",
  headers: {
    "Authorization": "Bearer {{ api_key }}",
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    platform: "X",
    caption: "Automated tweet scheduled via FeedBridge agent mode! 𝕏"
  })
});

const data = await response.json();
console.log(data);
